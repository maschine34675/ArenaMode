public static class NativeStopProbe
{
    static int checks;
    public static void Check(bool ok, string why) { checks++; if (!ok) throw new Exception(why); }
    static LocalGame Fresh()
    {
        ArenaSession.IsActive = false; ArenaRaidExit.Tick();
        ArenaSession.IsActive = ArenaSession.Exact = true;
        ArenaRespawnDriver.StandDown = false;
        Redeploy.Core.Engine.RespawnEngineState.IsPipelineRunning = false;
        var game = new LocalGame();
        var world = new GameWorld { MainPlayer = game.LocalPlayer };
        game.GameWorld = world; game.Owner = new EftGamePlayerOwner { Player = game.LocalPlayer };
        Singleton<AbstractGame>.Instance = game; Singleton<GameWorld>.Instance = world;
        ArenaSession.StartedGame = game; ArenaSession.StartedWorld = world;
        MonoBehaviourSingleton<EFT.UI.PreloaderUI>.Instance = new EFT.UI.PreloaderUI();
        PitFireTeamCompat.Reset(game.LocalPlayer, world);
        PitFireTeamRoundLifecycle.ResetProbe();
        ArenaMatchService.ResetProbe();
        ArenaMode.Plugin.ModLogger.Errors.Clear(); ArenaMode.Plugin.ModLogger.Warnings.Clear();
        return game;
    }
    public static int Run()
    {
        var game = Fresh();
        Check(PitFireTeamCompat.TryGetLiveFollowers(out var followers, out var problem) && followers.Count == 1, "valid live roster accepted");
        PitFireTeamCompat.Wrappers.Clear();
        Check(!PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem) && problem.Contains("living bound"), "live missing follower still fails closed");
        Check(ArenaMode.Plugin.ModLogger.Errors.Count == 1, "live violation still logs once");
        var latched = problem;
        game.Status = GameStatus.Stopping;
        Check(!PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem) && problem == latched, "native Stop must preserve an already latched fault");
        Check(ArenaMode.Plugin.ModLogger.Errors.Count == 1, "latched fault not repeated");

        foreach (var status in new[] { GameStatus.Stopping, GameStatus.Stopped })
        {
            game = Fresh(); game.Status = status; PitFireTeamCompat.Wrappers.Clear();
            Check(PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem) && followers.Count == 0 && problem == null,
                "stopped disposal must not create a roster fault");
            Check(PitFireTeamCompat.CurrentBodiesByAccountId.Count == 1 && PitFireTeamCompat.DiscoveryReads == 0
                && ArenaMode.Plugin.ModLogger.Errors.Count == 0, "stop retains identities and avoids discovery");
            ArenaMatchService.Tick();
            Check(ArenaMatchService.Current.Ticks == 0 && ArenaMatchService.ScoreboardReads == 0, "stopped match must not advance or sync");
            PitFireTeamRoundLifecycle.Enqueue(Thread.CurrentThread.ManagedThreadId);
            PitFireTeamCompat.Tick();
            Check(PitFireTeamRoundLifecycle.Completions == 1 && PitFireTeamRoundLifecycle.PendingCount == 0,
                "pending completion must drain during committed Stop");
            Check(PitFireTeamCompat.DiscoveryReads == 0 && PitFireTeamCompat.CurrentBodiesByAccountId.Count == 1, "drain does not clear roster or resume discovery");
        }

        game = Fresh(); game.Before = () => throw new Exception("native precommit failure");
        ArenaRespawnDriver.StandDown = true; ArenaRaidExit.Request(game);
        Check(game.Status == GameStatus.Started && !ArenaRaidExit.NativeStopOwnsCurrentRaid, "failed Stop remains a live raid");
        PitFireTeamCompat.Wrappers.Clear();
        Check(!PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem), "failed Stop does not hide a live roster fault");
        ArenaMatchService.Tick();
        Check(ArenaMatchService.Current.Ticks == 1 && ArenaMatchService.ScoreboardReads == 1, "failed Stop keeps match and scoreboard checks");

        game = Fresh(); Redeploy.Core.Engine.RespawnEngineState.IsPipelineRunning = true; ArenaRaidExit.Request(game);
        Check(!ArenaRaidExit.NativeStopOwnsCurrentRaid, "queued exit is not native commit");
        PitFireTeamCompat.Wrappers.Clear();
        Check(!PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem), "precommit queue does not hide live faults");

        foreach (var invalidate in new Action<LocalGame>[] {
            g => ArenaSession.Exact = false,
            g => g.GameWorld = new GameWorld(),
            g => Singleton<AbstractGame>.Instance = new LocalGame { Status = GameStatus.Stopping },
            g => Singleton<GameWorld>.Instance = new GameWorld() })
        {
            game = Fresh(); game.Status = GameStatus.Stopping; invalidate(game); PitFireTeamCompat.Wrappers.Clear();
            Check(!PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem), "foreign stop cannot mask the bound roster fault");
            ArenaMatchService.Tick();
            Check(ArenaMatchService.Current.Ticks == 1, "foreign stop does not own current match progression");
        }
        game = Fresh();
        ArenaMatchService.Current.DuringTick = () => { game.Status = GameStatus.Stopping; PitFireTeamCompat.Wrappers.Clear(); };
        ArenaMatchService.Tick();
        Check(ArenaMatchService.Current.Ticks == 1 && ArenaMatchService.ScoreboardReads == 0, "same-frame committed Stop must skip scoreboard");

        game = Fresh(); game.Status = GameStatus.Stopping; ArenaMatchService.TeamStartAbortPending = true;
        ArenaMatchService.Tick(); Check(ArenaMatchService.AbortTicks == 1, "failed-start owner still receives its tick");

        game = Fresh(); ArenaSession.IsActive = false; PitFireTeamCompat.LiveTeamArena = false;
        PitFireTeamCompat.Wrappers.Clear();
        Check(PitFireTeamCompat.TryGetLiveFollowers(out followers, out problem) && problem == null && PitFireTeamCompat.DiscoveryReads == 0, "ordinary raid does not inspect PFT roster");
        PitFireTeamRoundLifecycle.Enqueue(Thread.CurrentThread.ManagedThreadId);
        PitFireTeamCompat.Tick(); Check(PitFireTeamRoundLifecycle.Completions == 1, "old pending creator still drains outside Arena");
        PitFireTeamRoundLifecycle.Enqueue(Thread.CurrentThread.ManagedThreadId + 100);
        PitFireTeamCompat.Tick(); Check(PitFireTeamRoundLifecycle.PendingCount == 1, "foreign-thread completion remains queued");
        return checks;
    }
}

    {
        static readonly object StateLock = new object();

        static bool _roundTransitionActive, _followerDiscoveryFailureLogged;
        static PitFireTeamContractState _contractState;
        static Player _currentLocalPlayer;
        static object _currentLeaderWrapper;
        static int _frozenCompanionCount;
        static MethodInfo _getFollowersMethod = typeof(PitFireTeamCompat).GetMethod(nameof(GetWrappers));
        static MethodInfo _getFollowerBotMethod = typeof(Follower).GetMethod(nameof(Follower.GetBot));
        static MethodInfo _getFollowerBossMethod = typeof(Follower).GetMethod(nameof(Follower.GetBoss));
        static FieldInfo _bossRealPlayerField = typeof(Boss).GetField(nameof(Boss.realPlayer));
        public static readonly Dictionary<string, BotOwner> CurrentBodiesByAccountId = new Dictionary<string, BotOwner>();

        public static bool LiveTeamArena;
        public static bool RoundTransitionActive => _roundTransitionActive;
        public static int DiscoveryReads;
        public static IEnumerable GetWrappers() { DiscoveryReads++; return Wrappers; }
        static bool ShouldDeferInitialRosterDiscovery() => false;
        static bool TryGetCurrentProfileId(string account, out string id) { id = account == "ally-account" ? "ally-profile" : null; return id != null; }
        public static void Reset(Player leader, GameWorld world)
        {
            _runtimeRosterProblem = null; _roundTransitionActive = _followerDiscoveryFailureLogged = false;
            _currentLocalPlayer = leader; _currentLocalProfileId = leader.ProfileId;
            var boss = new Boss { realPlayer = leader }; _currentLeaderWrapper = boss;
            var bot = new BotOwner { ProfileId = "ally-profile" };
            CurrentBodiesByAccountId.Clear(); CurrentBodiesByAccountId["ally-account"] = bot;
            world.Registry[bot.ProfileId] = bot;
            Wrappers.Clear(); Wrappers.Add(new Follower { Bot = bot, Boss = boss });
            _frozenCompanionCount = 1; LiveTeamArena = true; DiscoveryReads = 0;
            _contractState = PitFireTeamContractState.Compatible;
        }
    }

        static readonly Queue<PendingRawPlayerCompletion> PendingRawPlayerCompletions = new Queue<PendingRawPlayerCompletion>();

        public static void ResetProbe() { PendingRawPlayerCompletions.Clear(); Completions = 0; }
        static void CompleteLeasedRawLocalPlayerTaskSafely(object state, object player, Exception failure) => Completions++;
